       ```swift
       import Observation

       @MainActor
       @Observable
       final class PublicProfileViewModel {

           enum ViewState: Equatable {
               case loading
               case loaded(displayName: String?, avatarColorHex: String?)
               case error(message: String)
           }

           var state: ViewState = .loading

           var isLoading: Bool {
               if case .loading = state { return true }
               return false
           }

           /// Closure override for unit testing. When non-nil, used instead of ProfileSharingService.
           /// Follows the fake-injection pattern from NotificationSchedulerTests.
           var fetchOverride: ((String) async throws -> (String?, String?))? = nil

           func fetchProfile(recordID: String) {
               state = .loading
               Task {
                   do {
                       let result: (displayName: String?, avatarColorHex: String?)
                       if let override = fetchOverride {
                           let (d, a) = try await override(recordID)
                           result = (displayName: d, avatarColorHex: a)
                       } else {
                           result = try await ProfileSharingService.fetchProfile(recordID: recordID)
                       }
                       state = .loaded(displayName: result.displayName, avatarColorHex: result.avatarColorHex)
                   } catch let error as CKError {
                       switch error.code {
                       case .unknownItem:
                           state = .error(message: "This profile is no longer available.")
                       case .networkFailure, .networkUnavailable:
                           state = .error(message: "Couldn't load profile. Check your internet connection and try again.")
                       default:
                           state = .error(message: "Couldn't load profile. Check your internet connection and try again.")
                       }
                   } catch {
                       state = .error(message: "Couldn't load profile. Check your internet connection and try again.")
                   }
               }
           }
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| External URL -> VitaminGApp.onOpenURL | Untrusted URL from any app enters the process; parsed by DeepLinkParser |
| DeepLinkParser -> AppRouter | Validated recordID string sets navigation state |
| AppRouter -> ContentView sheet | pendingPublicProfileRecordID triggers sheet presentation |
| PublicProfileViewModel -> CloudKit | recordID from URL used to fetch public record |
| CloudKit response -> PublicProfileView | displayName rendered in SwiftUI Text (plain text, no HTML) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-10-01 | Tampering | DeepLinkParser.recordID(from:) | mitigate | Guard: scheme == "vitaming", host == "profile", non-empty recordID. Only first path component used. Tested with 6 unit tests for malformed inputs. |
| T-10-02 | Information Disclosure | ProfileSharingService.fetchProfile | accept | CloudKit public database is intentionally public. Only displayName and avatarColorHex exposed -- user opted in via Phase 7 publish. |
| T-10-03 | Spoofing | vitaming:// URL scheme | accept | Custom scheme can be claimed by another app. Universal Links mitigate but are out of scope per D-09. Same risk profile as Phase 7 send-side. |
| T-10-04 | Tampering | PublicProfileView Text(displayName) | mitigate | SwiftUI Text() renders plain text, not HTML -- no injection. InputSanitizer.sanitizeForPublic was applied at publish time (Phase 7, T-07-08). |
| T-10-05 | Denial of Service | Repeated deep link opens | accept | Each tap = one CloudKit fetch. CloudKit rate-limits at server side. No local amplification. |
| T-10-06 | Repudiation | Deep link usage logging | accept | No logging of which profiles are viewed -- consistent with privacy-first app design. No audit requirement for public reads. |
</threat_model>
