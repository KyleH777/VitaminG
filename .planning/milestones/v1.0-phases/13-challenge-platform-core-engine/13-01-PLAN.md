```swift
import SwiftData
import Foundation

// MARK: - SchemaV4
//
// Version 4 of the Vitamin G data schema.
// Adds:
//   - ChallengeTemplate model — configurable challenge definitions (CHAL-01)
//   - UserChallenge model — user's instance of a challenge (CHAL-02)
//   - CheckIn model — daily check-in entry per UserChallenge (CHAL-03)
//
// CRITICAL: All V4 model types MUST live inside this enum.
// Goal, CompletionEvent, UserProfile, DailyWin are unchanged — referenced from V2/V3.
// All properties optional or defaulted — CloudKit sync compatibility (CLAUDE.md).
// No @Attribute(.unique) — CloudKit does not support atomic uniqueness.
// Both sides of every @Relationship declare `inverse:`.

enum SchemaV4: VersionedSchema {
    static var versionIdentifier = Schema.Version(4, 0, 0)

    static var models: [any PersistentModel.Type] {
        // V3 models unchanged + 3 new V4 types
        [SchemaV2.Goal.self,
         SchemaV2.CompletionEvent.self,
         SchemaV2.UserProfile.self,
         SchemaV3.DailyWin.self,
         SchemaV4.ChallengeTemplate.self,
         SchemaV4.UserChallenge.self,
         SchemaV4.CheckIn.self]
    }

    // MARK: - ChallengeTemplate (NEW in V4 — CHAL-01)
    @Model
    final class ChallengeTemplate {
        var id: UUID = UUID()
        var title: String?
        var challengeDescription: String?
        var category: String?               // "fitness" | "finance" | "sobriety"
        var challengeType: String?          // "featured" | "custom"
        var checkInType: String?            // "boolean" | "numeric" | "multiStep"
        var goalType: String?               // "streak" | "target" | "dateBound"
        var durationDays: Int?
        var milestonesJSON: String?         // JSON-encoded [MilestoneConfig]
        var accentColorHex: String?
        var iconName: String?
        var isFeatured: Bool = false
        var activeFrom: Date?
        var activeUntil: Date?
        var communitySize: Int = 0          // seeded value, not live count in Phase 13

        @Relationship(deleteRule: .nullify, inverse: \UserChallenge.template)
        var userChallenges: [UserChallenge]?

        init() {}
    }

    // MARK: - UserChallenge (NEW in V4 — CHAL-02)
    @Model
    final class UserChallenge {
        var id: UUID = UUID()
        var startDate: Date?
        var targetEndDate: Date?
        var currentStreak: Int = 0
        var longestStreak: Int = 0
        var totalCheckIns: Int = 0
        var statusRaw: String?              // "active" | "completed" | "abandoned"
        var milestoneHistoryJSON: String?   // JSON-encoded [Int] thresholds awarded
        var reminderHour: Int?
        var reminderMinute: Int?

        var template: ChallengeTemplate?

        @Relationship(deleteRule: .cascade, inverse: \CheckIn.userChallenge)
        var checkIns: [CheckIn]?

        init() {}
    }

    // MARK: - CheckIn (NEW in V4 — CHAL-03)
    @Model
    final class CheckIn {
        var id: UUID = UUID()
        var date: Date?
        var payloadBool: Bool?
        var payloadNumber: Double?
        var payloadNote: String?
        var timestamp: Date?

        var userChallenge: UserChallenge?

        init() {}
    }
}

// MARK: - Typealiases (V4)
typealias ChallengeTemplate = SchemaV4.ChallengeTemplate
typealias UserChallenge = SchemaV4.UserChallenge
typealias CheckIn = SchemaV4.CheckIn
```

Notes:
- All three classes use empty `init()` — properties are set after creation by callers (template static constants in Plan 02; CheckIn properties set in recordCheckIn).

- `template` and `userChallenge` are inverse relationships — declared on the "many" side without explicit `@Relationship` macro (the `@Relationship` on the "one" side declares the inverse via `\UserChallenge.template` keypath).
- Do NOT add Hashable conformance — `@Model` classes already conform to Identifiable/Hashable via `id`. If compile fails, Plan 03 will add explicit `Hashable` conformance via extension.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Disk-stored SwiftData store -> in-memory model graph | Migration runs on opaque on-disk file; corrupted/manipulated store flows back into app memory |
| App Group container -> Widget process | Both processes open the same store with the same schema; mismatch enables silent data loss |
| iCloud private DB -> Main app | New record types must be registered in CloudKit schema before sync; mis-registration drops records silently |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-13-01 | Tampering | SchemaV4 model properties | mitigate | All properties optional or defaulted (CloudKit safety); no `@Attribute(.unique)`; @Relationship inverses on both sides — enforced via grep acceptance criteria in Task 1 |
| T-13-02 | Denial of Service | Migration crash on launch | mitigate | Use `MigrationStage.lightweight` (purely additive — no transform code paths); both makeContainer and makeWidgetContainer updated together (Pitfall 1, 2) |
| T-13-03 | Information Disclosure | Widget process opens stale schema | mitigate | makeWidgetContainer also updated to SchemaV4 (Pitfall 2 acceptance criterion: `grep -c "SchemaV4.models"` returns 2) |
| T-13-04 | Tampering | CloudKit schema lacks new record types | mitigate | DEBUG `initializeCloudKitSchema` updated to include ChallengeTemplate/UserChallenge/CheckIn (Pitfall 3) |
| T-13-05 | Tampering | Concurrent migration + widget read | accept | Standard SwiftData behavior; widget retries on next timeline refresh; not Phase 13's concern |
