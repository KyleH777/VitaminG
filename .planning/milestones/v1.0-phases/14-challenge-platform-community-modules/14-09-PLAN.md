    import Foundation

    extension ChallengeTemplate {

        // MARK: - Module identifiers (Phase 14)
        enum ModuleIdentifier: String, CaseIterable {
            case spendingFreeze       = "spendingFreeze"
            case cravingTools         = "cravingTools"
            case transformationPhotos = "transformationPhotos"
            case nutritionLog         = "nutritionLog"
            case buddyAccountability  = "buddyAccountability"

            var displayName: String {
                switch self {
                case .spendingFreeze:       return "Spending Freeze"
                case .cravingTools:         return "Craving Tools"
                case .transformationPhotos: return "Transformation Photos"
                case .nutritionLog:         return "Nutrition Log"
                case .buddyAccountability:  return "Buddy Accountability"
                }
            }

            var systemImage: String {
                switch self {
                case .spendingFreeze:       return "snowflake"
                case .cravingTools:         return "wind"
                case .transformationPhotos: return "photo.stack.fill"
                case .nutritionLog:         return "fork.knife"
                case .buddyAccountability:  return "person.2.fill"
                }
            }
        }

        /// Decoded list of enabled module identifiers — empty if JSON is missing or malformed.
        /// Never throws — try? with empty-array fallback (RESEARCH.md anti-pattern: do not crash on decode failure).
        var enabledModules: [ModuleIdentifier] {
            guard let json = enabledModulesJSON,
                  let data = json.data(using: .utf8),
                  let decoded = try? JSONDecoder().decode([String].self, from: data) else {
                return []
            }
            return decoded.compactMap { ModuleIdentifier(rawValue: $0) }
        }

        func isModuleEnabled(_ module: ModuleIdentifier) -> Bool {
            enabledModules.contains(module)
        }

        // MARK: - Encode helper (used by CustomChallengeBuilderView)
        static func encodeModules(_ modules: [ModuleIdentifier]) -> String? {
            let raws = modules.map { $0.rawValue }
            guard let data = try? JSONEncoder().encode(raws),
                  let json = String(data: data, encoding: .utf8) else { return nil }
            return json
        }

        // MARK: - Privacy
        var isPrivate: Bool { privacy == nil || privacy == "private" }
        var isCommunity: Bool { privacy == "community" }
    }

    import SwiftUI
    import SwiftData

    struct CustomChallengeBuilderView: View {
        @Environment(\.dismiss) private var dismiss
        @Environment(\.modelContext) private var modelContext

        // Step state
        @State private var currentStep: Int = 1

        // Step 1
        @State private var name: String = ""
        @State private var category: String = "fitness"
        @State private var isCommunity: Bool = false
        @State private var accentColorHex: String = "#C4673A"  // terra default

        // Step 2
        @State private var checkInType: String = "boolean"
        @State private var goalType: String = "streak"
        @State private var streakDays: Int = 30
        @State private var targetValueText: String = ""
        @State private var targetUnit: String = ""
        @State private var endDate: Date = Calendar.current.date(byAdding: .day, value: 30, to: Date()) ?? Date()
        @State private var durationDays: Int = 30
        @State private var selectedModules: Set<ChallengeTemplate.ModuleIdentifier> = []

        // Validation
        @State private var validationError: String? = nil

        private static let categories: [(label: String, value: String)] = [
            ("Fitness", "fitness"),
            ("Finance", "finance"),
            ("Sobriety", "sobriety"),
            ("Mindfulness", "mindfulness"),
            ("Custom", "custom")
        ]
        private static let accentSwatches: [(name: String, hex: String)] = [
            ("Terra", "#C4673A"),
            ("Sage", "#7A9B76"),
            ("Gold", "#C9A24F"),
            ("Purple", "#7E5BAA"),
            ("Blue",   "#4A90E2")
        ]

        private var trimmedName: String { name.trimmingCharacters(in: .whitespacesAndNewlines) }
        private var isStep1Valid: Bool { !trimmedName.isEmpty }
        private var isStep2Valid: Bool {
            switch goalType {
            case "streak":    return true
            case "target":    return !targetValueText.trimmingCharacters(in: .whitespaces).isEmpty
                                  && Double(targetValueText) != nil
            case "dateBound": return endDate > Date()
            default:          return false
            }
        }

        private var accentColor: Color { Color(hex: accentColorHex) }

        var body: some View {
            NavigationStack {
                Form {
                    Section {
                        Text("Step \(currentStep) of 2")
                            .font(.caption.weight(.regular))
                            .fontDesign(.rounded)
                            .foregroundStyle(VGTheme.muted)
                            .frame(maxWidth: .infinity, alignment: .center)
                    }
                    if currentStep == 1 {
                        step1Sections
                    } else {
                        step2Sections
                    }

                    if let error = validationError {
                        Text(error)
                            .font(.caption).fontDesign(.rounded)
                            .foregroundStyle(.red)
                    }

                    if currentStep == 2 {
                        Section {
                            Button("Create Challenge") { create() }
                                .font(.body.weight(.semibold)).fontDesign(.rounded)
                                .frame(maxWidth: .infinity, minHeight: 44)
                                .background(accentColor)
                                .foregroundStyle(.white)
                                .clipShape(RoundedRectangle(cornerRadius: 12))
                                .listRowInsets(EdgeInsets())
                                .listRowBackground(Color.clear)
                                .disabled(!isStep1Valid || !isStep2Valid)
                        }
                    }
                }
                .scrollContentBackground(.hidden)
                .background(VGTheme.sandLight)
                .navigationTitle("Build Your Challenge")
                .navigationBarTitleDisplayMode(.inline)
                .toolbar {
                    ToolbarItem(placement: .navigationBarLeading) {
                        Button("Discard Builder") { dismiss() }
                            .font(.body).fontDesign(.rounded)
                    }
                    ToolbarItem(placement: .navigationBarTrailing) {
                        if currentStep == 1 {
                            Button("Next") {
                                if isStep1Valid {
                                    validationError = nil
                                    currentStep = 2
                                } else {
                                    validationError = "Please enter a challenge name."
                                }
                            }
                            .font(.body.weight(.semibold)).fontDesign(.rounded)
                            .disabled(!isStep1Valid)
                        } else {
                            Button("Back") {
                                validationError = nil
                                currentStep = 1
                            }
                            .font(.body).fontDesign(.rounded)
                        }
                    }
                }
            }
        }

        // MARK: - Step 1
        @ViewBuilder
        private var step1Sections: some View {
            Section("The Basics") {
                TextField("e.g., Daily Meditation, No Sugar Month", text: $name)
                    .font(.body).fontDesign(.rounded)
                    .onChange(of: name) { _, new in
                        if new.count > 60 { name = String(new.prefix(60)) }
                    }
                Picker("Category", selection: $category) {
                    ForEach(Self.categories, id: \.value) { item in
                        Text(item.label).tag(item.value)
                    }
                }
                .pickerStyle(.menu)
                Toggle(isOn: $isCommunity) {
                    Text(isCommunity ? "Community" : "Just Me")
                        .font(.body).fontDesign(.rounded)
                }
                HStack(spacing: 12) {
                    Text("Accent Color")
                        .font(.body).fontDesign(.rounded)
                        .foregroundStyle(VGTheme.clay)
                    Spacer()
                    ForEach(Self.accentSwatches, id: \.hex) { swatch in
                        Button {
                            accentColorHex = swatch.hex
                        } label: {
                            Circle()
                                .fill(Color(hex: swatch.hex))
                                .frame(width: 32, height: 32)
                                .overlay(
                                    Circle()
                                        .strokeBorder(
                                            accentColorHex == swatch.hex ? VGTheme.clay : Color.clear,
                                            lineWidth: 2
                                        )
                                )
                                .accessibilityLabel("\(swatch.name)\(accentColorHex == swatch.hex ? " selected" : "")")
                        }
                        .buttonStyle(.plain)
                    }
                }
            }
        }

        // MARK: - Step 2
        @ViewBuilder
        private var step2Sections: some View {
            Section("Check-In & Goal") {
                Picker("Check-In Type", selection: $checkInType) {
                    Text("Daily Yes/No").tag("boolean")
                    Text("Enter a Number").tag("numeric")
                    Text("Multi-Step").tag("multiStep")
                }
                .pickerStyle(.segmented)

                Picker("Goal Type", selection: $goalType) {
                    Text("Build a Streak").tag("streak")
                    Text("Reach a Target").tag("target")
                    Text("End by a Date").tag("dateBound")
                }
                .pickerStyle(.segmented)

                switch goalType {
                case "streak":
                    Stepper("Target Days: \(streakDays)", value: $streakDays, in: 7...365)
                        .font(.body).fontDesign(.rounded)
                case "target":
                    TextField("Goal Value", text: $targetValueText)
                        .keyboardType(.decimalPad)
                        .font(.body).fontDesign(.rounded)
                    TextField("Unit (e.g. $, km, lbs)", text: $targetUnit)
                        .font(.body).fontDesign(.rounded)
                case "dateBound":
                    DatePicker("End Date", selection: $endDate, in: Date()..., displayedComponents: [.date])
                        .font(.body).fontDesign(.rounded)
                default: EmptyView()
                }

                Stepper("Duration (days): \(durationDays)", value: $durationDays, in: 7...365)
                    .font(.body).fontDesign(.rounded)
            }
        }

        // MARK: - Create
        private func create() {
            // Final validation
            if !isStep1Valid {
                validationError = "Please enter a challenge name."
                currentStep = 1
                return
            }
            if !isStep2Valid {
                switch goalType {
                case "target":    validationError = "Please enter a goal value."
                case "dateBound": validationError = "End date must be in the future."
                default:          validationError = "Please complete the goal details."
                }
                return
            }

            let template = ChallengeTemplate()
            template.id = UUID()
            template.title = InputSanitizer.sanitize(trimmedName)
            template.subtitle = nil
            template.category = category
            template.challengeType = "custom"
            template.checkInType = checkInType
            template.goalType = goalType
            template.goalValue = (goalType == "target") ? Double(targetValueText) : nil
            template.goalUnit = (goalType == "target") ? InputSanitizer.sanitize(targetUnit) : nil
            template.endDate = (goalType == "dateBound") ? endDate : nil
            template.durationDays = (goalType == "streak") ? streakDays : durationDays
            template.accentColorHex = accentColorHex
            template.iconSymbol = "star.fill"
            template.isFeatured = false
            template.isActive = true
            template.privacy = isCommunity ? "community" : "private"
            // Encode selected module identifiers — e.g. ["spendingFreeze","cravingTools"]
            let moduleRawValues = selectedModules.map(\.rawValue)
            template.enabledModulesJSON = (try? JSONEncoder().encode(moduleRawValues)).flatMap { String(data: $0, encoding: .utf8) }
            template.milestonesJSON = makeDefaultMilestonesJSON()
            modelContext.insert(template)
            try? modelContext.save()
            dismiss()
        }

        /// Default milestone thresholds for streak challenges: 7, 30, 90, durationDays.
        /// Encoded as a JSON [String : Any]-shape array — same structure as featured templates.
        private func makeDefaultMilestonesJSON() -> String? {
            guard goalType == "streak" else { return nil }
            let baseDays = [7, 30, 90, streakDays].sorted()
            // De-duplicate
            var seen = Set<Int>()
            let dedup = baseDays.filter { seen.insert($0).inserted }
            // Match the milestonesJSON shape used by ChallengeTemplate+Featured.swift
            // (executor MUST mirror the existing encoded shape — read that file before writing)
            struct MilestonePayload: Codable {
                let day: Int
                let message: String
            }
            let payloads = dedup.map { day in
                MilestonePayload(day: day, message: "Day \(day) — keep going!")
            }
            guard let data = try? JSONEncoder().encode(payloads),
                  let json = String(data: data, encoding: .utf8) else { return nil }
            return json
        }
    }
