```swift
import Foundation

enum VGQuoteScheduler {
    private static let shuffleKey = "vg_quoteShuffleYear"
    private static let orderKey   = "vg_quoteShuffleOrder"

    static func todayQuote() -> VGQuote {
        let cal = Calendar.current
        let year = cal.component(.year, from: Date())
        let dayOfYear = cal.ordinality(of: .day, in: .year, for: Date()) ?? 1

        if UserDefaults.standard.integer(forKey: shuffleKey) != year
            || UserDefaults.standard.array(forKey: orderKey) == nil {
            regenerateShuffle(for: year)
        }

        let order = UserDefaults.standard.array(forKey: orderKey) as? [Int] ?? []
        let count = VGQuoteBank.all.count
        let index = order[(dayOfYear - 1) % max(order.count, 1)] % count
        return VGQuoteBank.all[index]
    }

    private static func regenerateShuffle(for year: Int) {
        var indices = Array(0..<VGQuoteBank.all.count)
        var rng = SeededRNG(seed: UInt64(year))
        for i in stride(from: indices.count - 1, through: 1, by: -1) {
            let j = Int(rng.next() % UInt64(i + 1))
            indices.swapAt(i, j)
        }
        UserDefaults.standard.set(year, forKey: shuffleKey)
        UserDefaults.standard.set(indices, forKey: orderKey)
    }
}

struct SeededRNG {
    var state: UInt64
    init(seed: UInt64) { state = seed }
    mutating func next() -> UInt64 {
        state = state &* 6364136223846793005 &+ 1442695040888963407
        return state
    }
}
```

```swift
import Foundation
import FirebaseAuth

enum VerificationStep: Equatable {
    case phoneEntry
    case codeEntry
}

@Observable
final class PhoneVerificationViewModel {
    var step: VerificationStep = .phoneEntry
    var phoneNumber: String = ""
    var verificationCode: String = ""
    var verificationID: String? = nil
    var isLoading: Bool = false
    var errorMessage: String? = nil

    var isCodeValid: Bool {
        verificationCode.count == 6 && verificationCode.allSatisfy(\.isNumber)
    }

    func sendCode(dialCode: String) async {
        let fullNumber = "+\(dialCode)\(phoneNumber)"
        await MainActor.run { isLoading = true; errorMessage = nil }
        do {
            let id = try await PhoneAuthProvider.provider().verifyPhoneNumber(fullNumber, uiDelegate: nil)
            await MainActor.run {
                verificationID = id
                step = .codeEntry
                isLoading = false
            }
        } catch {
            await MainActor.run {
                errorMessage = error.localizedDescription
                isLoading = false
            }
        }
    }

    func verifyCode() async -> Bool {
        guard let id = verificationID, isCodeValid else {
            await MainActor.run { errorMessage = "Enter a valid 6-digit code." }
            return false
        }
        await MainActor.run { isLoading = true; errorMessage = nil }
        let credential = PhoneAuthProvider.provider().credential(withVerificationID: id, verificationCode: verificationCode)
        do {
            try await Auth.auth().signIn(with: credential)
            await MainActor.run { isLoading = false }
            return true
        } catch {
            await MainActor.run {
                errorMessage = error.localizedDescription
                isLoading = false
            }
            return false
        }
    }
}
```

```swift
import Foundation
import StoreKit

enum PurchaseState: Equatable {
    case idle
    case loading(String)    // product ID being purchased
    case success
    case failed(String)     // error description

    static func == (lhs: PurchaseState, rhs: PurchaseState) -> Bool {
        switch (lhs, rhs) {
        case (.idle, .idle), (.success, .success): return true
        case (.loading(let a), .loading(let b)): return a == b
        case (.failed(let a), .failed(let b)): return a == b
        default: return false
        }
    }
}

@Observable
final class TipViewModel {
    static let productIDs = [
        "com.kyleharrington.VitaminG.tip_small",
        "com.kyleharrington.VitaminG.tip_medium",
        "com.kyleharrington.VitaminG.tip_large"
    ]

    var products: [Product] = []
    var purchaseState: PurchaseState = .idle

    func loadProducts() async {
        do {
            products = try await Product.products(for: Self.productIDs)
        } catch {
            purchaseState = .failed(error.localizedDescription)
        }
    }

    func purchase(_ product: Product) async {
        purchaseState = .loading(product.id)
        do {
            let result = try await product.purchase()
            switch result {
            case .success(let verification):
                switch verification {
                case .verified:
                    purchaseState = .success
                case .unverified(_, let error):
                    purchaseState = .failed(error.localizedDescription)
                }
            case .userCancelled:
                purchaseState = .idle
            case .pending:
                purchaseState = .idle
            @unknown default:
                purchaseState = .idle
            }
        } catch {
            purchaseState = .failed(error.localizedDescription)
        }
    }
}
```

| Old | New | Condition |

| `.background(VGTheme.sandLight` | `.background(VGTheme.background` | All uses as a background |
